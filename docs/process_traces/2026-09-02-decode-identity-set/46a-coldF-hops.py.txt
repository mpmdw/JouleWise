tmp = pathlib.Path(tempfile.mkdtemp(prefix="coldgate45-hops-"))
gone_root = tmp / "gone"          # never created
BOOT = str(uuid.uuid4())
Z = "0" * 64

def probe(name, fn):
    try:
        out = fn(); print(f"{name:<44} NO ERROR -> {out!r}"[:200])
    except LaunchLineageError as e:
        print(f"{name:<44} reason_code={e.reason_code!r} :: {str(e)[:110]}")
    except Exception as e:
        print(f"{name:<44} {type(e).__name__}: {str(e)[:110]}")

print("== A. the five hops named by the packet, each executed on a nonexistent path")
probe("H1 consumption receipt gone", lambda: ar._read_v2_consumption(
    gone_root / "pack" / "arm_readiness.consumptions" / "arm-0001.consumed.json"))

# H2 pack root gone: run the REAL _replay_consumed_arm body; only the arm-receipt reader is
# stubbed so control reaches the pack-root resolve at the `arm["pack"]["pack_root"]` line.
cons_path = tmp / "pack-a" / "arm_readiness.consumptions" / "arm-0001.consumed.json"
arm_stub = {"receipt_id": "arm-0001", "receipt_kind": "arm",
            "pack": {"pack_root": str(gone_root / "pack-a"), "pack_id": "pack-a"}}
consumption_stub = {"arm_receipt": {"path": "arm_readiness.receipts/arm-0001.json",
                                    "sha256": Z, "receipt_id": "arm-0001"}}
def h2():
    with mock.patch.object(ar, "_read_arm_with_sidecar", return_value=(arm_stub, b"", Z)):
        return ar._replay_consumed_arm(None, consumption_stub, cons_path,

            "arm_context": arm_ctx, "reviewed_main": {"head_commit": "a" * 40},
            "valid_until_monotonic_ns": 10**18, "pack": {"pack_root": str(tmp / "pack-a")}}

    "boot_session_id": BOOT, "pack_id": "pack-a", "pack_sha256": Z, "plan_id": "plan-1",

def manifest_bytes(window_plan_root):
    return render_json({"schema_version": ar.LAUNCH_MANIFEST_SCHEMA, "boot_session_id": BOOT,
        "window_plan_root": str(window_plan_root),

    "pack_id": "pack-a", "plan_id": "plan-1", "window_id": "win-1", "bracket_session_id": "bracket-1",

    "settle": {"path": str(gone_root / "c-1.settle.json"), "sha256": Z}, "completion": None}

         mock.patch.object(ar, "_read_exact_launch_reference",
