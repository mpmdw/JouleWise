import re, json, sys
SRC = "/Users/edr/code/JouleWise-wt-p3main-opus0928/docs/paper/draft-v2-skeleton.md"
text = open(SRC, encoding="utf-8").read()
lines = text.split("\n")
# mask HTML comments (keep offsets)
def mask(pattern, t, flags=re.DOTALL):
    return re.sub(pattern, lambda m: re.sub(r"[^\n]", "\x00", m.group(0)), t, flags=flags)
masked = mask(r"<!--.*?-->", text)
# references section: lines 923..951 (## 9. References to before Appendix A)
ref_start = text.index("## 9. References"); ref_end = text.index("## Appendix A.")
masked = masked[:ref_start] + re.sub(r"[^\n]", "\x00", masked[ref_start:ref_end]) + masked[ref_end:]
# headings: mask the heading number prefix only
masked = re.sub(r"(?m)^(#+ )((?:Appendix )?[A-Z]?\d+(?:\.\d+)*\.?)", lambda m: m.group(1)+"\x00"*len(m.group(2)), masked)
# URLs, markdown link targets
masked = mask(r"https?://\S+", masked)
masked = mask(r"\]\([^)]*\)", masked)
# IDs: D-123, A-12, DG-071, DX-001, PE-01, SYN-01, C-028, [FILL:..], r03 etc, hex hashes, run ids
masked = mask(r"\b[A-Z][A-Za-z]*-\d+(?:[–/-]\d+)*\b", masked)
masked = mask(r"\b(?:Section|Sections|Figure|Figures|Table|Tables|Appendix|Appendices|Algorithm|Step|step|Equation|Eq\.)\s+[A-Z]?\d+(?:\.\d+)*(?:[–-][A-Z]?\d+(?:\.\d+)*)?", masked)
masked = mask(r"\b(?:January|February|March|April|May|June|July|August|September|October|November|December)(?:\s+\d{1,2},)?\s+\d{4}\b", masked)
masked = mask(r"\b\d+(?:\.\d+)?B\b", masked)
masked = mask(r"\\[A-Za-z]+", masked)
masked = mask(r"\b[A-Za-z_][\w./-]*\d[\w./-]*", masked)
masked = mask(r"\b(?=[0-9a-f]*[a-f])(?=[0-9a-f]*\d)[0-9a-f]{7,64}\b", masked)
masked = mask(r"\b\d{8}T\d{6}-[0-9a-f]{8}\b", masked)
# citation markers [12], [3, 4], [5–7]
masked = mask(r"\[\d+(?:\s*[,–-]\s*\d+)*\]", masked)
NUM = re.compile(r"(?<![\w.\\])[+\-−±]?(?:(?:\d{1,3}(?:,\d{3})+(?!\d)|\d+)(?:\.\d+)?|\.\d+)(?:[eE][+\-−]?\d+)?%?(?![\w])(?!\.\d)")
masked = re.sub(r"(?m)^(\s*)(\d+)(\. )", lambda m: m.group(1)+"\x00"*len(m.group(2))+m.group(3), masked)
out = []
for m in NUM.finditer(masked):
    s = m.group(0)
    if not re.search(r"\d", s): continue
    # strip trailing punctuation dot/comma
    tok = s.rstrip(".,")
    start = m.start(); end = start + len(tok)
    line = text.count("\n", 0, start) + 1
    ls = text.rfind("\n", 0, start) + 1
    le = text.find("\n", start); le = len(text) if le < 0 else le
    out.append(dict(i=len(out), start=start, end=end, line=line, lit=tok, ctx=text[max(ls,start-40):min(le,end+30)]))
json.dump(out, open("literals.json","w"), indent=1, ensure_ascii=False)
print(len(out))
