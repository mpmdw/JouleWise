def _cached_night(n=5, width=1, mode="registered", policy=None, budget_j=None):
    if policy is not None:
        return _original_night(n, width, mode, policy, budget_j)
    key=(n,width,mode,budget_j)

        _night_cache[key]=_original_night(n,width,mode,policy,budget_j)
    g, reg, roster, predictions = _night_cache[key]
    return copy.deepcopy(g), reg, copy.deepcopy(roster), copy.deepcopy(predictions)

class Replace(ast.NodeTransformer):
    def __init__(self, target, replacement):
        self.target=target; self.replacement=replacement
    def generic_visit(self,node):
        if node is self.target:
            return ast.copy_location(self.replacement,node)
        return super().generic_visit(node)

def mutant_source(d):
    tree=copy.deepcopy(TREE)
    if d['index'] is not None:
        node=list(ast.walk(tree))[d['index']]
        if d['kind']=='comparison':
            op=node.ops[d['operand']]
            node.ops[d['operand']]={ast.Gt:ast.GtE,ast.GtE:ast.Gt,ast.Lt:ast.LtE,ast.LtE:ast.Lt}[type(op)]()
        elif d['kind']=='bool_operand':
            vals=[v for j,v in enumerate(node.values) if j!=d['operand']]
            replacement=vals[0] if len(vals)==1 else ast.BoolOp(op=node.op,values=vals)
            tree=Replace(node,replacement).visit(tree)
        elif d['kind']=='guard':
            tree=Replace(node,ast.Pass()).visit(tree)
        elif d['kind']=='collapse':
            tree=Replace(node,node.args[d['operand']]).visit(tree)
    elif d['kind']=='fraction_literal':
        node=next(n for n in ast.walk(tree) if isinstance(n,ast.Attribute) and n.lineno==280 and n.attr=='CAP_BOUND_FRACTION')
        tree=Replace(node,ast.Constant(.2)).visit(tree)
    elif d['kind']=='drop_pairing':
        node=next(n for n in ast.walk(tree) if isinstance(n,ast.ListComp) and n.lineno==258)
        node.generators[0].ifs=[]
    elif d['kind'] in ('parent_energy_per_item','cell_energy_per_item'):
        lineno=d['line']
        node=next(n for n in ast.walk(tree) if isinstance(n,ast.Subscript) and n.lineno==lineno and isinstance(n.value,ast.Subscript) and isinstance(n.value.value,ast.Name) and n.value.value.id=='window_by_key' and isinstance(n.slice,ast.Constant) and n.slice.value=='gross_j')
        keyvar='k' if lineno==247 else 'key'
        replacement=ast.BinOp(left=copy.deepcopy(node),op=ast.Mult(),right=ast.Call(func=ast.Name(id='len',ctx=ast.Load()),args=[ast.Subscript(value=ast.Subscript(value=ast.Name(id='blocks',ctx=ast.Load()),slice=ast.Subscript(value=ast.Name(id=keyvar,ctx=ast.Load()),slice=ast.Constant(0),ctx=ast.Load()),ctx=ast.Load()),slice=ast.Constant('items'),ctx=ast.Load())],keywords=[]))
        tree=Replace(node,replacement).visit(tree)
    elif d['kind']=='bound_literal':
        node=next(n for n in ast.walk(tree) if isinstance(n,ast.Assign) and len(n.targets)==1 and isinstance(n.targets[0],ast.Name) and n.targets[0].id==d['constant'])
        node.value=ast.Constant(d['value'])
    return ast.unparse(ast.fix_missing_locations(tree))+'\n'

def load_module(path):
    name='joulewise.scored_reduce'
    spec=importlib.util.spec_from_file_location(name,path)
    mod=importlib.util.module_from_spec(spec)
    sys.modules[name]=mod
    spec.loader.exec_module(mod)
    return mod

def run_suite(methods):
    suite=unittest.TestSuite(ScoredReduceTests(m) for m in methods)
    out=io.StringIO()
    result=unittest.TextTestRunner(stream=out,verbosity=0,failfast=True).run(suite)
    failed=(result.failures or result.errors)
    first=None
    if failed:
        first=dict(test=failed[0][0].id().split('.')[-1],type='failure' if result.failures else 'error',tail=failed[0][1].splitlines()[-5:])
    return dict(result='pass' if result.wasSuccessful() else 'fail',testsRun=result.testsRun,first=first,tail=out.getvalue().splitlines()[-4:])
