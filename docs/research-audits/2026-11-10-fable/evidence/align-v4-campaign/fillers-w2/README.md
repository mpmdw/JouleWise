Regenerate the queue files from the audit checkout root.

```sh
python3.13 -I -B - <<'PY'
import json
import sys
from pathlib import Path

campaign = Path("docs/research-audits/2026-11-10-fable/evidence/align-v4-campaign")
output = Path(sys.argv[1]) if len(sys.argv) > 1 else campaign / "fillers-w2"
expiry = "2026-12-08T12:35:29-08:00"


def write_queue(directory, name, queue):
    target = output / directory / name
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(json.dumps(queue, indent=1) + "\n", encoding="utf-8")


for source_dir in ("fillers-ready/queues", "f2-stab/qual-pin/queues"):
    for source in sorted((campaign / source_dir).glob("*.json")):
        queue = json.loads(source.read_bytes())
        queue["expires_at"] = expiry
        write_queue("queues-1leg", source.name, queue)

        if not source.stem.startswith("f2-inst-"):
            continue
        spans = queue["spans"]
        for suffix, offset in (("a", 0), ("b", 1)):
            pairs = []
            for first, second in zip(spans[offset::2], spans[offset + 1::2]):
                legs = first["legs"] + second["legs"]
                span_id = first["block_id"] + "-" + "-".join(leg.lower() for leg in legs)
                pairs.append(dict(first, span_id=span_id, legs=legs))
            write_queue("queues-2leg", f"{source.stem}-{suffix}.json", dict(queue, spans=pairs))
PY
```
