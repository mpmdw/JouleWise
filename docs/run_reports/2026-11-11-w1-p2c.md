## New test and mutations

Exactly one test method was added:
`RefusalAllowlistTests.test_v4_desk_borrows_only_the_site_free_meter_probe`
in `tests/hazards/test_refusal_allowlist.py`.

It parses every scanned Python module, including imports nested in functions,
and pins the names imported from `scripts.align_v4_session` to exactly
`{_probe_meter_live}`. Aliases retain their original imported name; module
imports, parent-package module imports, wildcard imports from that module and relative forms
cannot silently borrow the entire desk program. It parses the one top-level
probe definition, extracts **only its body**, and calls the registry's own
`census.scan_source` on that body, requiring an empty result. It imports or
executes no session-program code and never runs the meter probe.

The harness copied only scanned Python sources plus the session-program source
to a `TemporaryDirectory` under `/private/tmp` and patched the test's `ROOT`
to that copy. No production source was mutated. Results:

|---|---|
| Pristine sources | 1 test, 0 failures |
| Driver's existing `from scripts.align_v4_session import _probe_meter_live` gains `, _record_bound` | 1 test, 1 expected failure; `_record_bound` is an extra borrowed name |

on keeping that temporary file:

```python
import ast
import json
import shutil
import tempfile
import unittest
from pathlib import Path
from unittest.mock import patch
from tests.hazards import refusal_census as c
from tests.hazards import test_refusal_allowlist as t

doc = c.load_allowlist()
with tempfile.TemporaryDirectory(prefix='w1-p2c-pin-', dir='/private/tmp') as tmp:
    root = Path(tmp)
    for file in set(doc['scan']['modules']) | {'scripts/align_v4_session.py'}:
        if file.endswith('.py'):
            destination = root / file
            destination.parent.mkdir(parents=True, exist_ok=True)
            shutil.copyfile(t.ROOT / file, destination)

    def check(label, expected_failure, needle=''):
        case = t.RefusalAllowlistTests(method)
        case.document = doc
        result = unittest.TestResult()
        with patch.object(t, 'ROOT', root):
            case.run(result)
        assert not result.errors, result.errors
        assert len(result.failures) == int(expected_failure), result.failures
        if expected_failure:
            assert needle in result.failures[0][1], result.failures
        print(f'{label}: {result.testsRun} test; {len(result.failures)} expected failure(s)', flush=True)
        if expected_failure:
            print(result.failures[0][1].split('AssertionError:', 1)[1].strip()[:900], flush=True)

    check('pristine temporary copy', False)
    driver = root / 'joulewise/b5/driver.py'
    original = driver.read_text()
    anchor = 'from scripts.align_v4_session import _probe_meter_live'
    assert original.count(anchor) == 1
    driver.write_text(original.replace(anchor, anchor + ', _record_bound', 1))
    check('second imported name _record_bound', True, '_record_bound')
    driver.write_text(original)

    session = root / 'scripts/align_v4_session.py'
    original = session.read_text()
    probe = next(node for node in ast.parse(original).body
                 if isinstance(node, ast.FunctionDef) and node.name == '_probe_meter_live')
    lines = original.splitlines(keepends=True)
    lines.insert(probe.body[0].end_lineno, '    raise RuntimeError("RL-86 mutation")\n')
    session.write_text(''.join(lines))
    check('raise inside _probe_meter_live', True, 'RuntimeError')
    session.write_text(original)
    check('restored temporary copy', False)
```
